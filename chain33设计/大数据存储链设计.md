# Chain33 大数据存储链设计

## 1. 背景与动机

### 1.1 问题描述

现代零知识证明系统（如 STARKs）生成的证明数据通常较大：

| 证明类型 | 典型大小 | 说明 |
|----------|----------|------|
| SNARK (Groth16) | ~200 bytes | 最小，但需要可信设置 |
| PLONK | ~400-800 bytes | 通用可信设置 |
| **STARK** | **20-200 KB** | 无可信设置，透明性强 |
| FRI 证明 | 50-500 KB | 根据多项式度数变化 |

将大量 STARK 证明存储到主流区块链面临的问题：

```
┌─────────────────────────────────────────────────────────────────┐
│                    存储成本问题                                  │
├─────────────────────────────────────────────────────────────────┤
│                                                                 │
│  以太坊 Calldata 成本：                                         │
│  • 非零字节：16 gas/byte                                        │
│  • 100 KB 证明 ≈ 1,600,000 gas ≈ $50-100（高峰期更贵）         │
│                                                                 │
│  每日 1000 个证明 = $50,000-100,000/天                          │
│                                                                 │
│  这对于高频应用（游戏、DeFi、IoT）是不可接受的成本              │
│                                                                 │
└─────────────────────────────────────────────────────────────────┘
```

### 1.2 设计目标

将 Chain33 改造为专门优化大数据存储的区块链：

1. **低成本存储**：存储成本降低 100-1000 倍
2. **高吞吐量**：支持大量证明数据的快速写入
3. **数据可用性**：保证存储数据的持久性和可访问性
4. **与主链互操作**：可与以太坊等主链进行验证交互

```go
// 存储费用计算
type StoragePricing struct {
    // 基础费用（每字节）
    BasePerByte     uint64  // 基础存储费用
    
    // 时间因子
    PermanentMultiplier  float64  // 永久存储倍数（如 10x）
    YearlyRate           float64  // 年存储费率
    
    // 批量折扣
    BatchDiscount        float64  // 批量提交折扣
    
    // 数据类型折扣
    ProofDiscount        float64  // ZK 证明专项折扣
}

func CalculateFee(tx *DataStoreTx, pricing *StoragePricing) uint64 {
    size := len(tx.Data)
    baseFee := uint64(size) * pricing.BasePerByte
    
    // 应用折扣
    if tx.DataType == DataTypeSTARKProof {
        baseFee = uint64(float64(baseFee) * (1 - pricing.ProofDiscount))
    }
    
    // 时间因子
    if tx.ExpiryBlock == 0 {
        baseFee = uint64(float64(baseFee) * pricing.PermanentMultiplier)
    } else {
        blocks := tx.ExpiryBlock - currentBlock
        years := float64(blocks) / BlocksPerYear
        baseFee = uint64(float64(baseFee) * years * pricing.YearlyRate)
    }
    
    return baseFee
}
```

## 3. 交易大小限制与拆分策略

### 3.1 单笔交易大小分析

**问题**：单笔交易太大会导致以下问题：

```
┌─────────────────────────────────────────────────────────────────┐
│                  大交易的网络传播问题                            │
├─────────────────────────────────────────────────────────────────┤
│                                                                 │
│  1. P2P 广播延迟                                                │
│     • 大交易需要更长时间在节点间传播                           │
│     • 可能导致交易在共识轮次内无法到达所有验证者               │
│     • 增加孤块和分叉风险                                       │
│                                                                 │
│  2. 内存池压力                                                  │
│     • 大交易占用更多内存池空间                                 │
│     • 可能导致其他交易被拒绝                                   │
│     • DoS 攻击风险                                             │
│                                                                 │
│  3. 区块打包效率                                                │
│     • 单笔大交易可能无法与其他交易并行打包                     │
│     • 降低区块利用率                                           │
│                                                                 │
│  4. 验证延迟                                                    │
│     • 大交易验证时间更长                                       │
│     • 影响共识轮次时间                                         │
│                                                                 │
└─────────────────────────────────────────────────────────────────┘
```

### 3.2 交易大小限制（参考 Celestia）

**Celestia 的限制**：
- 单笔 Blob 最大：~2 MiB (1,973,786 bytes)
- 建议大小：< 500 KiB（确保快速打包）
- 区块大小：8 MB / 6 秒

**Chain33 存储链建议限制**：

| 参数 | 值 | 说明 |
|------|-----|------|
| 单笔交易推荐大小 | **256 KB** | 最佳网络传播效率 |
| 单笔交易最大大小 | **512 KB** | 确保广播可靠性 |
| 单笔交易硬上限 | **2 MB** | 极限情况，需更高 gas |
| 区块数据上限 | **8 MB** | 每区块最大数据量 |
| 区块时间 | **3-6 秒** | 共识轮次 |

### 3.3 大数据拆分策略

```go
// 数据拆分配置
const (
    RecommendedBlobSize = 256 * 1024  // 256 KB - 推荐大小
    MaxBlobSize         = 512 * 1024  // 512 KB - 最大大小
    HardLimitBlobSize   = 2 * 1024 * 1024 // 2 MB - 硬上限
)

// 大数据拆分结构
type SplitDataSubmission struct {
    // 原始数据标识
    OriginalDataID  [32]byte
    
    // 总分片数
    TotalChunks     uint32
    
    // 当前分片索引
    ChunkIndex      uint32
    
    // 分片数据
    ChunkData       []byte
    
    // 分片 Merkle 证明（证明属于原始数据）
    ChunkProof      []byte
    
    // 原始数据 Merkle 根
    OriginalRoot    [32]byte
}

// 自动拆分大数据
func SplitLargeData(data []byte, dataID [32]byte) ([]SplitDataSubmission, error) {
    if len(data) <= RecommendedBlobSize {
        // 无需拆分
        return nil, nil
    }
    
    // 计算分片数
    chunkSize := RecommendedBlobSize
    numChunks := (len(data) + chunkSize - 1) / chunkSize
    
    // 构建 Merkle 树
    chunks := splitIntoChunks(data, chunkSize)
    merkleTree := buildMerkleTree(chunks)
    
    submissions := make([]SplitDataSubmission, numChunks)
    for i, chunk := range chunks {
        submissions[i] = SplitDataSubmission{
            OriginalDataID: dataID,
            TotalChunks:    uint32(numChunks),
            ChunkIndex:     uint32(i),
            ChunkData:      chunk,
            ChunkProof:     merkleTree.GetProof(i),
            OriginalRoot:   merkleTree.Root(),
        }
    }
    
    return submissions, nil
}

// 重组分片数据
func ReassembleData(submissions []SplitDataSubmission) ([]byte, error) {
    // 验证所有分片属于同一原始数据
    if !validateChunks(submissions) {
        return nil, errors.New("invalid chunks")
    }
    
    // 按索引排序
    sort.Slice(submissions, func(i, j int) bool {
        return submissions[i].ChunkIndex < submissions[j].ChunkIndex
    })
    
    // 合并数据
    var result []byte
    for _, sub := range submissions {
        result = append(result, sub.ChunkData...)
    }
    
    return result, nil
}
```

### 3.4 Gas 费用与大小关系

```go
// Gas 费用随数据大小的计算（参考 Celestia）
type GasPricing struct {
    // 固定基础费用
    BaseFee         uint64  // 65,000 gas
    
    // 每字节 gas
    GasPerByte      uint64  // 1 gas/byte（基础）
    
    // 大交易惩罚系数（超过推荐大小后）
    LargeTxPenalty  float64 // 1.5x - 2x
}

func CalculateGas(dataSize int, pricing *GasPricing) uint64 {
    gas := pricing.BaseFee
    
    if dataSize <= RecommendedBlobSize {
        // 推荐大小内：线性计费
        gas += uint64(dataSize) * pricing.GasPerByte
    } else if dataSize <= MaxBlobSize {
        // 超过推荐但在最大范围内：1.5x 惩罚
        gas += uint64(RecommendedBlobSize) * pricing.GasPerByte
        excess := dataSize - RecommendedBlobSize
        gas += uint64(float64(excess) * float64(pricing.GasPerByte) * 1.5)
    } else if dataSize <= HardLimitBlobSize {
        // 超过最大但在硬上限内：2x 惩罚
        gas += uint64(RecommendedBlobSize) * pricing.GasPerByte
        gas += uint64(float64(MaxBlobSize-RecommendedBlobSize) * float64(pricing.GasPerByte) * 1.5)
        excess := dataSize - MaxBlobSize
        gas += uint64(float64(excess) * float64(pricing.GasPerByte) * 2.0)
    } else {
        return 0 // 拒绝超过硬上限的交易
    }
    
    return gas
}
```

## 4. 数据可用性层 (Data Availability Layer)

### 4.1 设计参考：Celestia

### 4.5 纠删码实现

```go
// Reed-Solomon 纠删码配置
type ErasureCodingConfig struct {
    DataShards   int  // 原始数据分片数
    ParityShards int  // 校验分片数
    ShardSize    int  // 每个分片大小
}

// 默认配置：50% 冗余，可容忍 50% 节点离线
var DefaultConfig = ErasureCodingConfig{
    DataShards:   64,
    ParityShards: 64,
    ShardSize:    512, // bytes
}

// 编码数据
func EncodeData(data []byte, config ErasureCodingConfig) ([][]byte, error) {
    enc, _ := reedsolomon.New(config.DataShards, config.ParityShards)
    
    // 分片
    shards, _ := enc.Split(data)
    
    // 生成校验分片
    enc.Encode(shards)
    
    return shards, nil
}

// 从任意 DataShards 个分片恢复数据
func RecoverData(shards [][]byte, config ErasureCodingConfig) ([]byte, error) {
    enc, _ := reedsolomon.New(config.DataShards, config.ParityShards)
    
    // 重建缺失分片
    enc.Reconstruct(shards)
    
    // 合并数据分片
    var buf bytes.Buffer
    enc.Join(&buf, shards, len(shards[0])*config.DataShards)
    
    return buf.Bytes(), nil
}
```

```go
        }
    }

}
```

### 6.1 数据承诺上链

```solidity
// Ethereum 上的数据可用性合约
contract DataAvailabilityBridge {
    // Chain33 存储链的数据承诺
    mapping(bytes32 => DataCommitment) public commitments;
    
    struct DataCommitment {
        bytes32 dataRoot;      // 数据 Merkle 根
        bytes32 kzgCommitment; // KZG 承诺
        uint256 blockHeight;   // Chain33 区块高度
        uint256 timestamp;
        bool verified;
    }
    
    // 提交数据承诺
    function submitCommitment(
        bytes32 dataId,
        bytes32 dataRoot,
        bytes32 kzgCommitment,
        uint256 chain33Block,
        bytes calldata proof
    ) external {
        // 验证 Chain33 区块证明
        require(verifyChain33BlockProof(chain33Block, proof), "Invalid block proof");
        
        commitments[dataId] = DataCommitment({
            dataRoot: dataRoot,
            kzgCommitment: kzgCommitment,
            blockHeight: chain33Block,
            timestamp: block.timestamp,
            verified: true
        });
        
        emit DataCommitmentSubmitted(dataId, dataRoot, chain33Block);
    }
    
    // 验证数据包含证明
    function verifyDataInclusion(
        bytes32 dataId,
        bytes calldata data,
        bytes32[] calldata merkleProof
    ) external view returns (bool) {
        DataCommitment memory commitment = commitments[dataId];
        require(commitment.verified, "Commitment not found");
        
        // 验证 Merkle 证明
        bytes32 dataHash = keccak256(data);
        return MerkleProof.verify(merkleProof, commitment.dataRoot, dataHash);
    }
}
```
